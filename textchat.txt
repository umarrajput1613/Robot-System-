CMakeLists.txt

```cmake
cmake_minimum_required(VERSION 3.20)
project(CodexCLI VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -Wall -Wextra")

# Find required packages
find_package(ftxui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(CURL REQUIRED)

# Include directories
include_directories(${CMAKE_SOURCE_DIR}/src)

# Source files
set(SOURCES
    src/main.cpp
    src/ui/interface.cpp
    src/core/app.cpp
    src/core/command_handler.cpp
    src/ai/ai_client.cpp
    src/ai/streaming_handler.cpp
    src/filesystem/explorer.cpp
    src/filesystem/file_watcher.cpp
    src/utils/spinner.cpp
    src/utils/syntax_highlighter.cpp
    src/utils/autocomplete.cpp
)

# Executable
add_executable(codex-cli ${SOURCES})

# Link libraries
target_link_libraries(codex-cli
    ftxui::screen
    ftxui::dom
    ftxui::component
    nlohmann_json::nlohmann_json
    CURL::libcurl
)

# Windows-specific settings
if(WIN32)
    target_compile_definitions(codex-cli PRIVATE _WIN32_WINNT=0x0601)
    target_link_libraries(codex-cli ws2_32)
endif()
```

```cpp
#include "interface.hpp"
#include "theme.hpp"
#include "../utils/syntax_highlighter.hpp"
#include "../utils/spinner.hpp"
#include <ftxui/dom/elements.hpp>
#include <ftxui/component/component.hpp>
#include <ftxui/component/event.hpp>
#include <algorithm>

namespace codex {
namespace ui {

using namespace ftxui;

Interface::Interface() 
    : screen_(ScreenInteractive::Fullscreen()),
      current_workspace_(std::filesystem::current_path().string()),
      show_loading_(false),
      selected_file_index_(0) {
}

Interface::~Interface() = default;

void Interface::Initialize() {
    BuildUI();
}

void Interface::BuildUI() {
    // Input component with history
    input_component_ = Input(&current_input_, "Type your command...");
    input_component_ |= CatchEvent([this](Event event) {
        if (event == Event::Return) {
            if (!current_input_.empty()) {
                HandleInput(current_input_);
                current_input_.clear();
            }
            return true;
        }
        return false;
    });
    
    // Sidebar component
    auto sidebar_container = Container::Vertical({});
    sidebar_component_ = Renderer(sidebar_container, [this] {
        Elements children;
        
        // Workspace header
        children.push_back(
            text(" EXPLORER ") | bold | color(Theme::Purple) | hcenter |
            ftxui::border | size(HEIGHT, EQUAL, 1)
        );
        
        children.push_back(separator());
        
        // File list
        for (size_t i = 0; i < file_list_.size(); ++i) {
            auto file_color = (i == selected_file_index_) ? Theme::Cyan : Color::White;
            std::string icon = (std::filesystem::is_directory(file_list_[i])) ? "📁 " : "📄 ";
            
            children.push_back(
                text(icon + std::filesystem::path(file_list_[i]).filename().string()) |
                color(file_color) |
                (i == selected_file_index_ ? inverted : nothing)
            );
        }
        
        return vbox({
            window(text(" EXPLORER ") | color(Theme::Purple) | bold,
                vbox(children) | flex),
        }) | size(WIDTH, EQUAL, 30) | bgcolor(Theme::Surface);
    });
    
    // Chat component
    chat_component_ = Renderer([this] {
        Elements messages;
        
        for (const auto& [role, content] : chat_history_) {
            messages.push_back(RenderChatMessage(role, content));
            messages.push_back(separator());
        }
        
        if (show_loading_) {
            messages.push_back(
                hbox({
                    text(" " + Spinner::GetFrame() + " ") | color(Theme::Cyan),
                    text(" AI is thinking...") | color(Theme::Purple),
                })
            );
        }
        
        return vbox(messages) | flex | yframe;
    });
    
    // Main container
    auto main_container = Container::Horizontal({
        sidebar_component_,
        Container::Vertical({
            chat_component_,
            input_component_,
        }),
    });
    
    // Status bar
    auto status_bar = Renderer([this] {
        std::string status = status_message_.empty() ? "Ready" : status_message_;
        return hbox({
            text(" " + status + " ") | color(Theme::Cyan),
            filler(),
            text(" Workspace: " + current_workspace_ + " ") | color(Theme::Purple),
        }) | bgcolor(Theme::Surface) | size(HEIGHT, EQUAL, 1);
    });
    
    // Main renderer
    main_component_ = Renderer(main_container, [this, status_bar] {
        return vbox({
            main_container->Render() | flex,
            separator(),
            status_bar->Render(),
        }) | Theme::GetStyle();
    });
}

void Interface::RenderChatMessage(const std::string& role, const std::string& content) {
    // This would be implemented to render markdown with syntax highlighting
    // For brevity, simplified version shown
    Elements lines;
    
    std::string prefix = (role == "user") ? "You" : "AI";
    Color roleColor = (role == "user") ? Theme::Cyan : Theme::Purple;
    
    lines.push_back(
        text(" " + prefix + " ") | bold | color(roleColor) | bgcolor(Theme::Surface)
    );
    
    // Parse content for markdown and code blocks
    // Syntax highlighting would be applied here
    
    return vbox(lines);
}

void Interface::HandleInput(const std::string& input) {
    AddChatMessage("user", input);
    // Command would be processed by app core
}

void Interface::AddChatMessage(const std::string& role, const std::string& content, bool isStreaming) {
    chat_history_.emplace_back(role, content);
    UpdateChatDisplay();
}

void Interface::UpdateChatDisplay() {
    screen_.PostEvent(Event::Custom);
}

void Interface::UpdateWorkspace(const std::string& path) {
    current_workspace_ = path;
}

void Interface::UpdateStatus(const std::string& status) {
    status_message_ = status;
}

void Interface::ShowLoading(bool show) {
    show_loading_ = show;
}

void Interface::UpdateFileList(const std::vector<std::string>& files) {
    file_list_ = files;
}

void Interface::Render() {
    screen_.Loop(main_component_);
}

} // namespace ui
} // namespace codex
```

```cpp
#pragma once
#include <string>
#include <chrono>

namespace codex {

class Spinner {
public:
    static std::string GetFrame() {
        static const char* frames[] = {"⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏"};
        static int frame_index = 0;
        
        auto now = std::chrono::steady_clock::now();
        static auto last_update = now;
        
        if (std::chrono::duration_cast<std::chrono::milliseconds>(now - last_update).count() > 80) {
            frame_index = (frame_index + 1) % 10;
            last_update = now;
        }
        
        return frames[frame_index];
    }
};

} // namespace codex
```

```cpp
            "help", "exit", "clear", "ls", "cd", "run", "compile",

        };

    std::vector<std::string> GetSuggestions(const std::string& prefix) {

        for (const auto& cmd : commands_) {
```

```cpp
#include "explorer.hpp"
#include <algorithm>

namespace codex {
namespace filesystem {

Explorer::Explorer() 
    : current_path_(std::filesystem::current_path().string()) {
}

Explorer::~Explorer() = default;

void Explorer::SetPath(const std::string& path) {
    if (std::filesystem::exists(path) && std::filesystem::is_directory(path)) {
        current_path_ = path;
    }
}

std::vector<std::string> Explorer::ListContents() {
    std::vector<std::string> contents;
    
    try {
        for (const auto& entry : std::filesystem::directory_iterator(current_path_)) {
            contents.push_back(entry.path().string());
        }
        std::sort(contents.begin(), contents.end());
    } catch (const std::exception& e) {
        // Handle error
    }
    
    return contents;
}

std::string Explorer::GetCurrentPath() const {
    return current_path_;
}

bool Explorer::CreateDirectory(const std::string& name) {
    try {
        return std::filesystem::create_directory(current_path_ + "/" + name);
    } catch (...) {
        return false;
    }
}

bool Explorer::DeleteFile(const std::string& path) {
    try {
        return std::filesystem::remove(path);
    } catch (...) {
        return false;
    }
}

} // namespace filesystem
} // namespace codex
```
