#include "streaming_handler.hpp"
#include <chrono>

namespace codex {
namespace ai {

StreamingHandler::StreamingHandler() 
    : streaming_(false), active_(true) {
}

StreamingHandler::~StreamingHandler() {
    active_ = false;
    cv_.notify_all();
    if (processing_thread_.joinable()) {
        processing_thread_.join();
    }
}

void StreamingHandler::StartStream(ChunkCallback on_chunk, CompleteCallback on_complete) {
    on_chunk_ = on_chunk;
    on_complete_ = on_complete;
    streaming_ = true;
    
    processing_thread_ = std::thread(&StreamingHandler::ProcessStream, this);
}

void StreamingHandler::PushChunk(const std::string& chunk) {
    std::lock_guard<std::mutex> lock(queue_mutex_);
    chunk_queue_.push(chunk);
    cv_.notify_one();
}

void StreamingHandler::EndStream() {
    streaming_ = false;
    cv_.notify_all();
}

bool StreamingHandler::IsStreaming() const {
    return streaming_;
}

void StreamingHandler::ProcessStream() {
    std::string complete_response;
    
    while (active_) {
        std::unique_lock<std::mutex> lock(queue_mutex_);
        cv_.wait(lock, [this] { 
            return !chunk_queue_.empty() || !streaming_ || !active_; 
        });
        
        while (!chunk_queue_.empty()) {
            std::string chunk = chunk_queue_.front();
            chunk_queue_.pop();
            lock.unlock();
            
            if (on_chunk_) {
                on_chunk_(chunk);
            }
            complete_response += chunk;
            
            lock.lock();
        }
        
        if (!streaming_ && chunk_queue_.empty()) {
            if (on_complete_) {
                on_complete_(complete_response);
            }
            break;
        }
    }
}

} // namespace ai
} // namespace codex

    auto args = ParseCommand(command_line);

    const std::string& command_name = args[0];

namespace filesystem {

FileWatcher::FileWatcher() 
    : interval_(std::chrono::milliseconds(1000)), running_(false) {
}

FileWatcher::~FileWatcher() {
    Stop();
}

void FileWatcher::Watch(const std::string& path, FileCallback callback) {
    watch_path_ = path;
    callback_ = callback;
    running_ = true;
    watch_thread_ = std::thread(&FileWatcher::Poll, this);
}

void FileWatcher::Stop() {
    running_ = false;
    if (watch_thread_.joinable()) {
        watch_thread_.join();
    }
}

void FileWatcher::SetInterval(std::chrono::milliseconds interval) {
    interval_ = interval;
}

void FileWatcher::Poll() {
    while (running_) {
        try {
            for (const auto& entry : std::filesystem::recursive_directory_iterator(watch_path_)) {
                if (std::filesystem::is_regular_file(entry.path())) {
                    std::string path = entry.path().string();
                    std::string current_hash = GetFileHash(path);
                    
                    auto it = file_hashes_.find(path);
                    if (it != file_hashes_.end()) {
                        if (it->second != current_hash) {
                            callback_(path, "modified");
                            it->second = current_hash;
                        }
                    } else {
                        file_hashes_[path] = current_hash;
                        callback_(path, "created");
                    }
                }
            }
        } catch (const std::exception& e) {
            // Handle error
        }
        
        std::this_thread::sleep_for(interval_);
    }
}

std::string FileWatcher::GetFileHash(const std::string& path) {
    std::ifstream file(path, std::ios::binary);
    if (!file.is_open()) return "";
    
    std::stringstream buffer;
    buffer << file.rdbuf();
    std::string content = buffer.str();
    
    // Simple hash for demo - use proper hashing in production
    std::hash<std::string> hasher;
    std::stringstream hash_stream;
    hash_stream << std::hex << hasher(content);
    
    return hash_stream.str();
}

} // namespace filesystem

Element MarkdownRenderer::Render(const std::string& markdown) {
    auto tokens = Tokenize(markdown);
    Elements elements;
    
    for (const auto& token : tokens) {
        elements.push_back(RenderToken(token));
    }
    
    return vbox(elements);
}

std::vector<MarkdownRenderer::Token> MarkdownRenderer::Tokenize(const std::string& markdown) {
    std::vector<Token> tokens;
    std::regex code_block_regex(R"(```(\w*)\n([\s\S]*?)```)");
    std::regex inline_code_regex(R"(`([^`]+)`)");
    std::regex bold_regex(R"(\*\*([^*]+)\*\*)");
    std::regex italic_regex(R"(\*([^*]+)\*)");
    std::regex header_regex(R"(^(#{1,6})\s+(.+)$)");
    
    std::string remaining = markdown;
    std::smatch match;
    
    // Parse code blocks first
    while (std::regex_search(remaining, match, code_block_regex)) {
        // Add text before code block
        if (match.prefix().length() > 0) {
            Token text_token;
            text_token.type = Token::TEXT;
            text_token.content = match.prefix();
            tokens.push_back(text_token);
        }
        
        // Add code block
        Token code_token;
        code_token.type = Token::CODE_BLOCK;
        code_token.language = match[1];
        code_token.content = match[2];
        tokens.push_back(code_token);
        
        remaining = match.suffix();
    }
    
    // Parse remaining text for inline elements
    if (!remaining.empty()) {
        Token text_token;
        text_token.type = Token::TEXT;
        text_token.content = remaining;
        tokens.push_back(text_token);
    }
    
    return tokens;
}

Element MarkdownRenderer::RenderToken(const Token& token) {
    switch (token.type) {
        case Token::CODE_BLOCK:
            return RenderCodeBlock(token.content, token.language);
        case Token::TEXT:
            return RenderText(token.content);
        default:
            return text(token.content);
    }
}

Element MarkdownRenderer::RenderCodeBlock(const std::string& code, const std::string& language) {
    std::string highlighted = SyntaxHighlighter::Highlight(code, language);
    
    // Split into lines for proper display
    std::vector<Element> lines;
    std::stringstream ss(code);
    std::string line;
    
    while (std::getline(ss, line)) {
        lines.push_back(text("  " + line) | color(Color::White));
    }
    
    return vbox({
        text(" " + (language.empty() ? "code" : language) + " ") 
            | color(Theme::Purple) | bgcolor(Theme::Surface) | bold,
        separator(),
        vbox(lines) | bgcolor(Theme::Background),
        separator(),
    }) | border;
}

Element MarkdownRenderer::RenderText(const std::string& text) {
    std::vector<Element> paragraphs;
    std::stringstream ss(text);
    std::string paragraph;
    
    while (std::getline(ss, paragraph, '\n')) {
        if (!paragraph.empty()) {
            paragraphs.push_back(text(paragraph) | color(Color::White));
        } else {
            paragraphs.push_back(emptyElement());
        }
    }
    
    return vbox(paragraphs);
}
